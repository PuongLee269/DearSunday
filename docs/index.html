Đồng bộ giữa các máy

Dữ liệu được lưu chung trong một Google Sheet, qua một Google Apps Script bạn tự triển khai (miễn phí, làm 1 lần). Xem hướng dẫn bên dưới để lấy URL Web App và đặt một mã bí mật. Chỉ cần dán 2 thứ này vào mỗi máy (điện thoại, máy tính) một lần.

Hướng dẫn tạo nơi lưu (làm 1 lần)
  1. Vào sheets.google.com, tạo một Google Sheet mới, đặt tên tuỳ ý (ví dụ "DearSunday Data").
  2. Vào menu Tiện ích mở rộng → Apps Script.
  3. Xoá hết code mẫu, dán đoạn code bên dưới vào, đổi SECRET thành một mã bí mật do bạn tự đặt.
  4. Bấm Triển khai → Triển khai mới. Chọn loại Web app. Mục "Người thực thi" chọn Tôi, mục "Ai có quyền truy cập" chọn Bất kỳ ai. Bấm Triển khai, cấp quyền khi được hỏi.
  5. Sao chép URL kết thúc bằng /exec, dán vào ô "URL Web App" bên dưới trên từng máy, cùng với mã bí mật bạn đã đặt.
var SECRET = "doi-ma-bi-mat-cua-ban";
var SHEET_NAME = "Data";

function getSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) sh = ss.insertSheet(SHEET_NAME);
  return sh;
}
function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  if (e.parameter.secret !== SECRET) return json_({ok:false,error:"Sai mã bí mật"});
  if (e.parameter.action === "pull") {
    var sh = getSheet_();
    var v = sh.getRange(1,1).getValue();
    return json_({ok:true, data: v ? JSON.parse(v) : null});
  }
  return json_({ok:false,error:"Không rõ yêu cầu"});
}
function doPost(e) {
  var q = e.parameter;
  if (q.secret !== SECRET) return json_({ok:false,error:"Sai mã bí mật"});
  if (q.action === "push") {
    var sh = getSheet_();
    sh.getRange(1,1).setValue(e.postData.contents);
    sh.getRange(1,2).setValue(new Date());
    return json_({ok:true});
  }
  return json_({ok:false,error:"Không rõ yêu cầu"});
}
